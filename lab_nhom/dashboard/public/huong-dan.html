Gửi mỗi 5 giây (chỗ if (now - lastMsg > 5000) của code mẫu) và gửi ngay khi LED hoặc nút đổi trạng thái. Số không để trong ngoặc kép: "temp": 26.4, không phải "temp": "26.4".

3. Topic phụ

TopicChiềuNội dung
vju/iot/nhom01/statusESP32 → brokerKết nối xong gửi online (retained). offline là Last Will: ESP32 mất điện thì broker tự báo
vju/iot/nhom01/outputbroker → ESP32on / off khi giảng viên bấm bóng LED trên dashboard. Bật/tắt LED xong gửi lại gói JSON

4. Code tham khảo

// Thư viện: PubSubClient, ArduinoJson, Adafruit AHTX0
void reconnect() {
  while (!client.connected()) {
    if (client.connect("nhom01-esp32",
          "vju/iot/nhom01/status", 1, true, "offline")) {
      client.publish("vju/iot/nhom01/status", "online", true);
      client.subscribe("vju/iot/nhom01/output");
    } else {
      delay(5000);   // in client.state() để biết lỗi
    }
  }
}
// Thay 2 lệnh publish esp32/temperature, esp32/humidity
sensors_event_t hum, temp;
aht.getEvent(&hum, &temp);
